# After creating this file, restart WSL: wsl --shutdown (from PowerShell)

[wsl2]
# Memory allocation (adjust based on your system RAM)
# For AI/LLM work, allocate at least 50% of total RAM
# Example: 32GB system → 16GB for WSL
memory=16GB

# CPU cores (adjust based on your system)
# For AI orchestrator, allocate most cores for parallel processing
# Example: 16 cores → allocate 12-14 to WSL
processors=12

# Swap space (for large model loading)
# Set to 50% of allocated memory
swap=8GB

# Swap file location (optional, default is in WSL virtual disk)
# swapFile=C:\\Users\\YourUsername\\wsl-swap.vhdx

# Enable localhost port forwarding (CRITICAL for MCP servers)
# Allows Windows to access WSL services on localhost
localhostForwarding=true

# Enable nested virtualization (if using VMs inside WSL)
# Useful for container-in-container scenarios
nestedVirtualization=true

# Kernel command line arguments
# These optimize for development workloads
kernelCommandLine=vsyscall=emulate

# Enable page reporting for better memory management
pageReporting=true

# GUI applications support (WSLg)
# Useful for GUI-based AI tools
guiApplications=true

# DNS tunneling (helps with VPN/Tailscale)
# Set to true if you experience DNS issues
dnsTunneling=true

# Network mode
# - NAT (default): WSL has separate IP, port forwarding via Windows
# - mirrored: WSL mirrors Windows network, better for complex networking
# - bridged: WSL gets IP from router, acts like physical machine
#
# For orchestrator with Tailscale/Cloudflare, try 'mirrored' for best results:
# networkingMode=mirrored

# Firewall settings (when using mirrored mode)
# firewall=true

# Auto-proxy (useful with VPNs)
# autoProxy=true

# ==============================================================================
# PERFORMANCE TUNING NOTES
# ==============================================================================
#
# 1. For LLM inference (large models):
#    memory=24GB or higher
#    processors=16 or more
#    swap=16GB
#
# 2. For Docker-heavy workloads:
#    memory=16GB minimum
#    Consider increasing swap
#
# 3. For MCP server orchestration:
#    localhostForwarding=true (CRITICAL)
#    networkingMode=mirrored (for complex networking)
#
# 4. Monitoring resource usage:
#    - Task Manager → Performance → WSL
#    - Inside WSL: htop, free -h, df -h
#
# 5. If experiencing slowdowns:
#    - Reduce memory/processors allocation
#    - Disable pageReporting
#    - Check Windows Defender exclusions
