# GPU Workers Hosts Configuration
#
# Format: worker_name:ip_address:ssh_user
# Lines starting with # are comments
#
# Example configurations:

# Local network workers
rtx5090-worker:192.168.1.100:admin
rtx3090-worker:192.168.1.101:admin
rtx3060-worker:192.168.1.102:admin

# Tailscale network workers (use Tailscale IPs)
# rtx5090-worker:100.64.1.100:admin
# rtx3090-worker:100.64.1.101:admin
# rtx3060-worker:100.64.1.102:admin

# Mixed environment
# production-gpu-1:192.168.1.100:root
# staging-gpu-2:192.168.1.101:ubuntu
# dev-gpu-3:192.168.1.102:developer

# Multiple GPUs on same host (different ports)
# multi-gpu-host-5090:192.168.1.200:admin
# multi-gpu-host-3090:192.168.1.200:admin

# Cloud instances
# aws-gpu-instance:54.123.45.67:ec2-user
# gcp-gpu-instance:35.123.45.67:gcpuser
# azure-gpu-instance:20.123.45.67:azureuser

# Notes:
# - Ensure SSH key-based authentication is set up
# - Workers should have sudo access without password for the user
# - Verify network connectivity before running setup
# - Use static IPs or configure DHCP reservations
